if(WIN32)
    # Every executable produced in the build tree must also be runnable outside
    # the compiler environment.  MinGW's libgcc has a transitive dependency on
    # libwinpthread, so copying only the directly imported DLL is insufficient.
    get_filename_component(KR_TOOLCHAIN_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set(KR_WINDOWS_RUNTIME_DLLS zlib1.dll libgcc_s_seh-1.dll libwinpthread-1.dll)
    foreach(KR_RUNTIME_DLL IN LISTS KR_WINDOWS_RUNTIME_DLLS)
        set(KR_RUNTIME_SOURCE "${KR_TOOLCHAIN_BIN}/${KR_RUNTIME_DLL}")
        if(NOT EXISTS "${KR_RUNTIME_SOURCE}")
            message(FATAL_ERROR "Required Windows runtime is missing: ${KR_RUNTIME_SOURCE}")
        endif()
        configure_file("${KR_RUNTIME_SOURCE}" "${CMAKE_BINARY_DIR}/${KR_RUNTIME_DLL}" COPYONLY)
    endforeach()
    function(kr_bundle_windows_runtime target_name)
        foreach(KR_RUNTIME_DLL IN LISTS KR_WINDOWS_RUNTIME_DLLS)
            add_custom_command(TARGET ${target_name} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "${KR_TOOLCHAIN_BIN}/${KR_RUNTIME_DLL}"
                    "$<TARGET_FILE_DIR:${target_name}>/${KR_RUNTIME_DLL}")
        endforeach()
    endfunction()

    set(KR_WINSDK_ROOT "C:/Program Files (x86)/Windows Kits/10")
    file(GLOB KR_WINSDK_CANDIDATES LIST_DIRECTORIES true "${KR_WINSDK_ROOT}/Include/10.*")
    list(SORT KR_WINSDK_CANDIDATES COMPARE NATURAL ORDER DESCENDING)
    if(NOT KR_WINSDK_CANDIDATES)
        message(FATAL_ERROR "A Windows 10/11 SDK with C++/WinRT headers is required")
    endif()
    list(GET KR_WINSDK_CANDIDATES 0 KR_WINSDK_INCLUDE)
    get_filename_component(KR_WINSDK_VERSION "${KR_WINSDK_INCLUDE}" NAME)
    if(NOT EXISTS "${KR_WINSDK_INCLUDE}/cppwinrt" OR NOT EXISTS "${KR_WINSDK_ROOT}/Lib/${KR_WINSDK_VERSION}/um/x64/WindowsApp.lib")
        message(FATAL_ERROR "Windows SDK ${KR_WINSDK_VERSION} is incomplete")
    endif()
    target_include_directories(mycelia_native PUBLIC "${KR_WINSDK_ROOT}/Include/${KR_WINSDK_VERSION}/cppwinrt")
    target_link_libraries(mycelia_native PUBLIC bcrypt crypt32 winhttp ole32 shell32
        "${KR_WINSDK_ROOT}/Lib/${KR_WINSDK_VERSION}/um/x64/WindowsApp.lib"
        "${KR_WINSDK_ROOT}/Lib/${KR_WINSDK_VERSION}/um/x64/mfplat.lib"
        "${KR_WINSDK_ROOT}/Lib/${KR_WINSDK_VERSION}/um/x64/mfreadwrite.lib")
    target_link_libraries(mycelia_native PUBLIC "${KR_WINSDK_ROOT}/Lib/${KR_WINSDK_VERSION}/um/x64/SAPI.lib")
else()
    find_package(OpenSSL REQUIRED)
    target_link_libraries(mycelia_native PUBLIC OpenSSL::Crypto)
endif()
